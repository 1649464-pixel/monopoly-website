const rooms = new Map(); app.use(express.static(path.join(__dirname, 'public'))); function randomRoomCode() { const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; let code = ''; for (let i = 0; i < 5; i += 1) { code += chars[Math.floor(Math.random() * chars.length)]; } return code; } function createPlayer(socketId, name, idx) { return { socketId, id: socketId, name, color: BOT_PLAYER_COLORS[idx % BOT_PLAYER_COLORS.length], money: 1500, position: 0, bankrupt: false, owned: [], inJail: 0 }; } function getRoomForSocket(socketId) { for (const room of rooms.values()) { if (room.players.some((player) => player.socketId === socketId)) { return room; } } return null; } function serializeRoom(room) { const board = room.board.map((tile) => { const owner = tile.ownerId ? room.players.find((player) => player.socketId === tile.ownerId) : null; return { ...tile, owner: owner ? owner.name : null, ownerId: tile.ownerId || null }; }); return { code: room.code, hostSocketId: room.hostSocketId, started: room.started, winner: room.winner ? room.winner.name : null, currentPlayerId: room.players[room.currentPlayerIndex] ? room.players[room.currentPlayerIndex].socketId : null, waitingForBuy: room.waitingForBuy, buyOption: room.buyOption !== null && room.buyOption !== undefined ? { id: room.buyOption, name: board[room.buyOption]?.name || '', price: board[room.buyOption]?.price || 0 } : null, lastRoll: room.lastRoll, log: room.log, players: room.players.map((player) => ({ ...player, owned: [...player.owned] })), board }; } function nextPlayerIndex(room, startIndex) { let idx = startIndex; let safety = 0; while (safety < room.players.length * 2) { idx = (idx + 1) % room.players.length; if (!room.players[idx].bankrupt) { return idx; } safety += 1; } return startIndex; } function endTurn(room) { if (!room.started || room.winner) { return; } room.waitingForBuy = false; room.buyOption = null; room.lastRoll = [null, null]; room.currentPlayerIndex = nextPlayerIndex(room, room.currentPlayerIndex); emitRoomState(room); } function checkWinner(room) { const alivePlayers = room.players.filter((player) => !player.bankrupt); if (alivePlayers.length === 1 && room.started) { room.winner = alivePlayers[0]; addLog(room, `${alivePlayers[0].name} wins the game!`); emitRoomState(room); } } function bankruptPlayer(room, player, amount, creditor) { player.bankrupt = true; room.board.forEach((tile) => { if (tile.ownerId === player.socketId) { tile.ownerId = null; } }); player.owned = []; if (creditor && creditor.socketId !== player.socketId && amount > 0) { creditor.money += amount; } addLog(room, `${player.name} went bankrupt and lost all properties.`); checkWinner(room); } function drawChanceCard(room, player) { const card = CHANCE_CARDS[Math.floor(Math.random() * CHANCE_CARDS.length)]; addLog(room, `${player.name} drew a Chance card: ${card.text}`); if (card.kind === 'advanceStart') { player.position = 0; player.money += 200; addLog(room, `${player.name} moved to Start and collected $200.`); return; } if (card.kind === 'money') { player.money += card.amount; return; } if (card.kind === 'jail') { player.position = 10; player.inJail = 1; addLog(room, `${player.name} went to jail.`); return; } if (card.kind === 'repair') { const repairCost = card.amount * player.owned.length; if (player.money < repairCost) { bankruptPlayer(room, player, repairCost, null); return; } player.money -= repairCost; addLog(room, `${player.name} paid $${repairCost} in repairs.`); } } function applyTileEffects(room, player, tile, rollTotal) { if (tile.type === 'property' || tile.type === 'utility') { if (!tile.ownerId) { if (player.money >= tile.price) { room.waitingForBuy = true; room.buyOption = tile.id; addLog(room, `${player.name} landed on ${tile.name}. They can buy it for $${tile.price}.`); return; } addLog(room, `${player.name} landed on ${tile.name}, but cannot afford it.`); return; } if (tile.ownerId !== player.socketId) { const owner = room.players.find((candidate) => candidate.socketId === tile.ownerId); if (!owner) { tile.ownerId = null; return; } const rent = tile.rent * (tile.type === 'utility' ? rollTotal : 1); if (player.money < rent) { addLog(room, `${player.name} could not pay $${rent} rent to ${owner.name} and went bankrupt.`); bankruptPlayer(room, player, rent, owner); } else { player.money -= rent; owner.money += rent; addLog(room, `${player.name} paid $${rent} rent to ${owner.name}.`); } } return; } if (tile.type === 'tax') { const amount = tile.amount; if (player.money < amount) { addLog(room, `${player.name} could not pay $${amount} in taxes and went bankrupt.`); bankruptPlayer(room, player, amount, null); } else { player.money -= amount; addLog(room, `${player.name} paid $${amount} in taxes.`); } return; } if (tile.type === 'chance') { drawChanceCard(room, player); return; } if (tile.type === 'start') { player.money += tile.reward; addLog(room, `${player.name} collected $${tile.reward} from Start.`); return; } if (tile.type === 'jail') { addLog(room, `${player.name} is just visiting jail.`); } } function handleMove(room, player, steps) { const previousPosition = player.position; const nextPosition = (player.position + steps) % room.board.length; if (nextPosition < previousPosition) { player.money += 200; addLog(room, `${player.name} passed Start and collected $200.`); } player.position = nextPosition; const tile = room.board[player.position]; applyTileEffects(room, player, tile, steps); } function getRoomCodeFromSocket(socket) { return Array.from(rooms.keys()).find((code) => { const room = rooms.get(code); return room && room.players.some((player) => player.socketId === socket.id); }); } function ensureRoomExists(roomCode) { if (!rooms.has(roomCode)) { return null; } return rooms.get(roomCode); } function broadcastRoomList() { const summary = Array.from(rooms.entries()).map(([code, room]) => ({ code, players: room.players.length, started: room.started, host: room.hostSocketId }));
